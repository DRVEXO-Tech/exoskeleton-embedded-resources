4.必要的函数需要有返回值，返回值的定义如下,返回错误时要返回负值，需要添加负号。返回为长度时需要正数。

```c
typedef enum
{
    RET_OK              =  0,    // 成功
    RET_ERROR           =  1,   // 通用错误
    RET_BUSY            =  2,   // 设备或资源忙
    RET_TIMEOUT         =  3,   // 超时
    RET_INVALID_PARAM   =  4,   // 参数错误
    RET_NULL_PTR        =  5,   // 空指针
    RET_NOT_READY       =  6,   // 设备未准备好
    RET_NOT_INIT        =  7,   // 未初始化
    RET_NO_MEM          =  8,   // 内存不足
    RET_NO_RESOURCE     =  9,   // 资源不足
    RET_NOT_SUPPORT     =  10,  // 不支持该功能
    RET_OUT_OF_RANGE    =  11,  // 超出范围
    RET_OVERFLOW        =  12,  // 溢出
    RET_UNDERFLOW       =  13,  // 下溢
    RET_CHECKSUM_ERR    =  14,  // 校验错误
    RET_CRC_ERR         =  15,  // CRC错误
    RET_COMM_ERR        =  16,  // 通信错误
    RET_HW_ERR          =  17,  // 硬件错误
    RET_STATE_ERR       =  18,  // 状态错误
    RET_LOCKED          =  19,  // 资源被锁定
    RET_RETRY           =  20   // 需要重试
} ret_en;
```
